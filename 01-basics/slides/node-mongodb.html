const server = http.createServer((req, res) => {
  if (req.method === "GET" && req.url === "/") {
    res.writeHead(200, {
      "Content-Type": "application/json",
    });
    res.end(JSON.stringify({ message: "Hello" }));
  } else {
    res.writeHead(404);
    res.end();
  }
});

server.listen(3000, () => {

מחברים את Node ל-MongoDB עם MongooseConnecting Node to MongoDB with Mongoose

npm i mongoose. ספרייה שנותנת חיבור, סכמה, ולידציה ומודלnpm i mongoose. A library that gives you a connection, schemas, validation and models

1. חיבור, פעם אחת בעליית השרת1. Connect once, when the server starts

// config/db.js
import mongoose from "mongoose";

export async function connectDB() {
  await mongoose.connect(
    process.env.MONGODB_URI
  );
  console.log("MongoDB connected");
}

// server.js
await connectDB();
app.listen(3000);

2. סכמה ומודל2. Schema and model

// models/Book.js
const bookSchema = new mongoose.Schema(
  {
    title: { type: String, required: true },
    year: { type: Number, min: 1450 },
    author: { type: String, required: true },
    tags: [String],
    inStock: { type: Boolean, default: true },
  },
  { timestamps: true }
);

export const Book =
  mongoose.model("Book", bookSchema);

Router נפרד לכל משאב, ו-error handler אחד לכל השרתA separate Router per resource, and one error handler for the whole server

routes/books.js: קריאהroutes/books.js: reading

import { Router } from "express";
import { Book } from "../models/Book.js";
const router = Router();

router.get("/", async (req, res) => {
  const books = await Book.find();
  res.json(books);
});

router.get("/:id", async (req, res) => {
  const book = await Book.findById(req.params.id);
  if (!book) return res.sendStatus(404);
  res.json(book);
});
