router.post("/register", async (req, res) => {
  const { name, password } = req.body ?? {};
  const email = normalizeEmail(req.body?.email);

  if (typeof name !== "string" || !name.trim())
    return res.status(400).json({ error: "Name is required" });
  if (!EMAIL_RE.test(email))
    return res.status(400).json({ error: "A valid email is required" });
  if (typeof password !== "string" || password.length < 8)
    return res.status(400).json({ error: "Password: 8+ characters" });

  if (await User.exists({ email }))
    return res.status(409).json({ error: "Email is already registered" });

  const passwordHash = await bcrypt.hash(password, 10);
  const user = await User.create({ name, email, passwordHash });

  setAuthCookie(res, user);   // logged in right after sign-up
  res.status(201).json({ user });
});

התחברותLogin

מוצאים את המשתמש, משווים את הסיסמה ל-hash, ומחזירים טוקןFind the user, compare the password to the hash, and return a token

router.post("/login", async (req, res) => {
  const email = normalizeEmail(req.body?.email);
  const { password } = req.body ?? {};

  if (!email || typeof password !== "string")
    return res.status(400).json({ error: "Email and password are required" });

  const user = await User.findOne({ email })
    .select("+passwordHash");

  const ok = user !== null &&
    (await bcrypt.compare(password, user.passwordHash));

  if (!ok)
    return res.status(401).json({ error: "Wrong email or password" });

  setAuthCookie(res, user);
  res.json({ user });
});

יצירה ובדיקה של טוקןSigning and verifying a token

שתי פונקציות בקובץ אחד, כדי שהסוד וזמן התפוגה יוגדרו במקום אחדTwo functions in one file, so the secret and the expiry live in one place

// src/lib/tokens.js
import jwt from "jsonwebtoken";

export function signToken(user) {
  return jwt.sign(
    { sub: user.id, role: user.role },  // payload
    process.env.JWT_SECRET,             // secret
    { expiresIn: "7d" }
  );
}

export function verifyToken(token) {
  return jwt.verify(token, process.env.JWT_SECRET);
  // returns the payload, or throws
}

Middleware שבודק מי מחוברA middleware that checks who’s logged in

שומר הסף של כל נתיב מוגן: מוצא טוקן, בודק אותו, ושם את המשתמש ב-req.userThe gatekeeper of every protected route: finds a token, checks it, and puts the user on req.user

export async function requireAuth(req, res, next) {
  // 1. Find the token: cookie first, then a Bearer header
  const header = req.get("authorization");
  const token = req.cookies?.token ??
    (header?.startsWith("Bearer ") ? header.slice(7) : null);
  if (!token)
    return res.status(401).json({ error: "Not logged in" });

  // 2. Check the signature and expiry
  let payload;
  try {
    payload = verifyToken(token);
  } catch {
    return res.status(401).json({ error: "Invalid or expired token" });
  }

  // 3. Load the user: they may have been deleted
  const user = await User.findById(payload.sub);
  if (!user)
    return res.status(401).json({ error: "User no longer exists" });

  req.user = user;
  next();
}

export function requireRole(role) {

    if (req.user?.role !== role)
